SELECT PRODUCT_NAME, CATEGORY, COUNT(DISTINCT TRANSACTION_ID) AS TXN_COUNT, SUM(QUANTITY) AS UNITS, SUM(GROSS_SALES_INR) AS GROSS_SALES_INR, SUM(DISCOUNT_AMOUNT_INR) AS DISCOUNT_INR, SUM(REVENUE_INR) AS REVENUE_INR, SUM(RETURN_AMOUNT_INR) AS RETURNS_INR, SUM(NET_REVENUE_INR) AS NET_REVENUE_INR, SUM(COST_OF_GOODS_INR) AS COGS_INR, SUM(SHIPPING_COST_INR) AS SHIPPING_INR, SUM(MARKETING_COST_INR) AS MARKETING_INR, SUM(CONTRIBUTION_INR) AS CONTRIBUTION_INR, DIV0NULL(SUM(CONTRIBUTION_INR), SUM(NET_REVENUE_INR)) * 100 AS MARGIN_PCT, DIV0NULL(SUM(NET_REVENUE_INR), COUNT(DISTINCT TRANSACTION_ID)) AS AOV_INR, DIV0NULL(SUM(REVENUE_INR), SUM(QUANTITY)) AS AVG_SELLING_PRICE_INR, DIV0NULL(SUM(DISCOUNT_AMOUNT_INR), SUM(GROSS_SALES_INR)) * 100 AS EFFECTIVE_DISCOUNT_PCT, COUNT(DISTINCT IFF(RETURNED, TRANSACTION_ID, NULL)) AS RETURNED_TXNS, DIV0NULL(COUNT(DISTINCT IFF(RETURNED, TRANSACTION_ID, NULL)), COUNT(DISTINCT TRANSACTION_ID)) * 100 AS RETURN_RATE_PCT, COUNT_IF(CONTRIBUTION_INR < 0) AS NEG_CONTRIBUTION_TXNS, COUNT_IF(CONTRIBUTION_INR IS NULL) AS MISSING_COST_ROWS FROM RETAIL_ANALYTICS.SALES.V_SALES_OVERVIEW WHERE TRANSACTION_DATE >= '2025-03-01' AND TRANSACTION_DATE <= '2025-09-30' AND REGION IN ('North', 'South') AND PROMOTION_TYPE IN ('No Promotion') GROUP BY PRODUCT_NAME, CATEGORY ORDER BY PRODUCT_NAME, CATEGORY;
WITH b AS (SELECT * FROM RETAIL_ANALYTICS.SALES.V_SALES_OVERVIEW WHERE TRANSACTION_DATE >= '2025-03-01' AND TRANSACTION_DATE <= '2025-09-30' AND REGION IN ('North', 'South') AND PROMOTION_TYPE IN ('No Promotion')), mid AS (SELECT DATEADD(DAY, DATEDIFF(DAY, MIN(TRANSACTION_DATE), MAX(TRANSACTION_DATE)) / 2, MIN(TRANSACTION_DATE)) m FROM b) SELECT PRODUCT_NAME, COUNT_IF(TRANSACTION_DATE <= m) AS TXNS_FIRST_HALF, COUNT_IF(TRANSACTION_DATE > m) AS TXNS_SECOND_HALF, DIV0NULL(COUNT_IF(RETURNED AND TRANSACTION_DATE <= m), COUNT_IF(TRANSACTION_DATE <= m)) * 100 AS RR_FIRST_HALF_PCT, DIV0NULL(COUNT_IF(RETURNED AND TRANSACTION_DATE > m), COUNT_IF(TRANSACTION_DATE > m)) * 100 AS RR_SECOND_HALF_PCT FROM b, mid GROUP BY PRODUCT_NAME;
SELECT DISCOUNT_PCT, COUNT(DISTINCT TRANSACTION_ID) AS TXN_COUNT, SUM(QUANTITY) AS UNITS, SUM(GROSS_SALES_INR) AS GROSS_SALES_INR, SUM(DISCOUNT_AMOUNT_INR) AS DISCOUNT_INR, SUM(REVENUE_INR) AS REVENUE_INR, SUM(RETURN_AMOUNT_INR) AS RETURNS_INR, SUM(NET_REVENUE_INR) AS NET_REVENUE_INR, SUM(COST_OF_GOODS_INR) AS COGS_INR, SUM(SHIPPING_COST_INR) AS SHIPPING_INR, SUM(MARKETING_COST_INR) AS MARKETING_INR, SUM(CONTRIBUTION_INR) AS CONTRIBUTION_INR, DIV0NULL(SUM(CONTRIBUTION_INR), SUM(NET_REVENUE_INR)) * 100 AS MARGIN_PCT, DIV0NULL(SUM(NET_REVENUE_INR), COUNT(DISTINCT TRANSACTION_ID)) AS AOV_INR, DIV0NULL(SUM(REVENUE_INR), SUM(QUANTITY)) AS AVG_SELLING_PRICE_INR, DIV0NULL(SUM(DISCOUNT_AMOUNT_INR), SUM(GROSS_SALES_INR)) * 100 AS EFFECTIVE_DISCOUNT_PCT, COUNT(DISTINCT IFF(RETURNED, TRANSACTION_ID, NULL)) AS RETURNED_TXNS, DIV0NULL(COUNT(DISTINCT IFF(RETURNED, TRANSACTION_ID, NULL)), COUNT(DISTINCT TRANSACTION_ID)) * 100 AS RETURN_RATE_PCT, COUNT_IF(CONTRIBUTION_INR < 0) AS NEG_CONTRIBUTION_TXNS, COUNT_IF(CONTRIBUTION_INR IS NULL) AS MISSING_COST_ROWS FROM RETAIL_ANALYTICS.SALES.V_SALES_OVERVIEW WHERE TRANSACTION_DATE >= '2025-03-01' AND TRANSACTION_DATE <= '2025-09-30' AND REGION IN ('North', 'South') AND PROMOTION_TYPE IN ('No Promotion') GROUP BY DISCOUNT_PCT ORDER BY DISCOUNT_PCT;
SELECT * EXCLUDE (UPLOAD_ID) FROM RETAIL_ANALYTICS.SALES.V_SALES_OVERVIEW WHERE TRANSACTION_DATE >= '2025-03-01' AND TRANSACTION_DATE <= '2025-09-30' AND REGION IN ('North', 'South') AND PROMOTION_TYPE IN ('No Promotion') ORDER BY TRANSACTION_DATE, TRANSACTION_ID LIMIT 5;
SELECT CHECKED_AT, CHECK_NAME, SEVERITY, STATUS, FAILED_ROWS, TOTAL_ROWS, ARRAY_TO_STRING(SAMPLE_IDS, ', ') AS SAMPLE_IDS, DETAILS FROM RETAIL_ANALYTICS.SALES.DATA_QUALITY_RESULTS WHERE CHECK_RUN_ID = (SELECT CHECK_RUN_ID FROM RETAIL_ANALYTICS.SALES.DATA_QUALITY_RESULTS ORDER BY CHECKED_AT DESC LIMIT 1) ORDER BY CASE STATUS WHEN 'FAIL' THEN 0 WHEN 'WARN' THEN 1 WHEN 'INFO' THEN 2 ELSE 3 END, CHECK_NAME;
SELECT 'Base table' AS SOURCE, COUNT(DISTINCT TRANSACTION_ID) TXN_COUNT, SUM(NET_REVENUE_INR) NET_REVENUE_INR, SUM(NET_REVENUE_INR - COST_OF_GOODS_INR - SHIPPING_COST_INR - MARKETING_COST_INR) CONTRIBUTION_INR FROM RETAIL_ANALYTICS.SALES.SALES_TRANSACTIONS UNION ALL SELECT 'V_MONTHLY_SALES', SUM(TXN_COUNT), SUM(NET_REVENUE_INR), SUM(CONTRIBUTION_INR) FROM RETAIL_ANALYTICS.SALES.V_MONTHLY_SALES UNION ALL SELECT 'V_CATEGORY_PROFITABILITY', SUM(TXN_COUNT), SUM(NET_REVENUE_INR), SUM(CONTRIBUTION_INR) FROM RETAIL_ANALYTICS.SALES.V_CATEGORY_PROFITABILITY UNION ALL SELECT 'V_REGIONAL_PERFORMANCE', SUM(TXN_COUNT), SUM(NET_REVENUE_INR), SUM(CONTRIBUTION_INR) FROM RETAIL_ANALYTICS.SALES.V_REGIONAL_PERFORMANCE
